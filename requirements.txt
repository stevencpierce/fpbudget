# ── Version policy (audit M4, 2026-07-20) ────────────────────────────────
# Every dependency carries a CEILING pin (< next breaking major) so a deploy
# can never silently pull an incompatible major release. No floors are set —
# whatever the build currently resolves stays valid, so this change cannot
# break an existing environment. tzdata stays unpinned (calendar data).
# To true-pin exact versions later: run `pip freeze` in a Render shell and
# commit it as requirements.lock, then install from that.

flask-login<1
flask-sqlalchemy<4
python-dotenv<2
gunicorn<24
psycopg2-binary<3
werkzeug<4
anthropic<2
weasyprint<70
Flask-Mail<1
twilio<10
tzdata
# Pinned BELOW 1.36 — botocore 1.36 introduced default flexible request
# checksums (CRC32) that Cloudflare R2 rejects, sending botocore into an
# infinite retry/recursion loop on every upload. The env-var + Config()
# suppression we tried (AWS_REQUEST_CHECKSUM_CALCULATION=when_required, etc.)
# was unreliable in practice. Easiest durable fix: stay on 1.35.x where
# checksums are off by default. Revisit when R2 supports flexible checksums.
boto3>=1.35,<1.36
botocore>=1.35,<1.36
dropbox<13
flask-socketio<6
# gevent: upper-bounded at <26 on 2026-04-17. gevent 26.x breaks our
# custom GeventWebSocketNoSSLWorker — the gunicorn worker hangs during
# module import and never binds its port. gevent-websocket (required
# for our socketio transport) has been unmaintained since 2017 and
# doesn't track gevent's API changes past v25. Stay in the 24.x /
# 25.x range until we either fork gevent-websocket or move to
# python-socketio's built-in transport.
#
# Lower bound 24.10 still applies — older versions monkey-patch
# ssl.SSLContext incorrectly on Python 3.12+, causing infinite recursion
# in `super(SSLContext, SSLContext).options.__set__` when any HTTPS
# client initializes.
gevent>=24.10,<26
gevent-websocket<1
